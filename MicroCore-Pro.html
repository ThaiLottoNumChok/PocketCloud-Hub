function runBenchmark() { document.getElementById('screen').innerHTML = "[Benchmark Started]"; logToScreen("กำลังตรวจวัดความเร็วอินเทอร์เน็ต & พลังชิป..."); let speed = 10; // ค่าเริ่มต้นสำรองปลอดภัย if (navigator.connection && typeof navigator.connection.downlink === 'number') { speed = navigator.connection.downlink || 10; } let simulatedGHz = (speed * 0.5).toFixed(2); document.getElementById('cpu-speed').innerText = simulatedGHz + " GHz"; logToScreen("ความเร็วเน็ตขับเคลื่อน: " + speed + " Mbps"); logToScreen("พลังประมวลผลชิปจำลอง: " + simulatedGHz + " GHz ผ่านฉลุย!"); }
